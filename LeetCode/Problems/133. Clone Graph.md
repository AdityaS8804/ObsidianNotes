- Problem : Given a node in a graph, clone the entire graph
#### DFS approach
- Intuition : 
	- Using DFS recursion : 
		- Clone the node
		- Store node in dictionary
		- Loop through neighbors
		- If any neighbor already has a clone, append that value to newNeighbors list
		- Else, append dfs(neighbor_node) to newNeighbors list
		- After loop, clone.neighbors=newNeighbors
	- Base case :
		- If given node is None(no object) -> return None
		- If given node is isolated without any neighbors -> return clone of that node alone

- Code
```python
"""
# Definition for a Node.
class Node:
    def __init__(self, val = 0, neighbors = None):
        self.val = val
        self.neighbors = neighbors if neighbors is not None else []
"""

from typing import Optional


class Solution:
    def cloneGraph(self, node: Optional["Node"]) -> Optional["Node"]:
        check = defaultdict(lambda: -1)

        def dfs(n):

            clone = Node(n.val, n.neighbors)
            check[n] = clone
            newNeighbors = []
            for i in n.neighbors:
                if check[i] == -1:
                    newNeighbors.append(dfs(i))
                else:
                    newNeighbors.append(check[i])
            clone.neighbors = newNeighbors
            return clone

        if not node:
            return
        if len(node.neighbors) == 0:
            return Node(node.val, node.neighbors)
        return dfs(node)
